local Gui = { labels = {} }

local function trimText(g)
    local t = tostring(g.Text or "")
    t = t:gsub("<[^>]+>", "")
    t = t:lower():gsub("^%s+", ""):gsub("%s+$", "")
    return t
end

local function isShown(g)
    local o = g
    while o and o ~= game do
        if o:IsA("GuiObject") and not o.Visible then return false end
        if o:IsA("ScreenGui") and not o.Enabled then return false end
        o = o.Parent
    end
    return true
end

local function trackGui(d)
    if d:IsA("TextLabel") or d:IsA("TextButton") then Gui.labels[d] = true end
end

function Gui.find(patterns, exact)
    for g in pairs(Gui.labels) do
        if not g.Parent then
            Gui.labels[g] = nil
        else
            local t = trimText(g)
            if t ~= "" then
                for _, p in ipairs(patterns) do
                    local hit
                    if exact then hit = (t == p) else hit = (t:find(p, 1, true) ~= nil) end
                    if hit and isShown(g) then return g, p end
                end
            end
        end
    end
    return nil
end

function Gui.tap(btn)
    if not VIM or not btn or not btn:IsA("GuiObject") then return end
    local sg = btn:FindFirstAncestorOfClass("ScreenGui")
    local inset = Vector2.new(0, 0)
    if sg and not sg.IgnoreGuiInset then inset = GuiService:GetGuiInset() end
    local c = btn.AbsolutePosition + btn.AbsoluteSize / 2 + inset
    pcall(function()
        VIM:SendMouseMoveEvent(c.X, c.Y, game)
        task.wait(jitter(0.05, 0.1))
        VIM:SendMouseButtonEvent(c.X, c.Y, 0, true, game, 0)
        task.wait(jitter(0.04, 0.08))
        VIM:SendMouseButtonEvent(c.X, c.Y, 0, false, game, 0)
    end)
end

local Input = {}

local SIGNALS = { "MouseButton1Down", "MouseButton1Up", "MouseButton1Click", "Activated" }
function Input.fire(btn)
    if not btn then return false end
    if firesignal then
        for _, n in ipairs(SIGNALS) do pcall(function() firesignal(btn[n]) end) end
        return true
    end
    if getconnections then
        for _, n in ipairs(SIGNALS) do
            pcall(function()
                for _, cn in ipairs(getconnections(btn[n])) do cn:Fire() end
            end)
        end
        return true
    end
    return false
end

function Input.key(kc, hold)
    hold = hold or jitter(0.04, 0.09)
    local ok = false
    if VIM then ok = pcall(function() VIM:SendKeyEvent(true, kc, false, game) end) end
    if ok then
        task.wait(hold)
        pcall(function() VIM:SendKeyEvent(false, kc, false, game) end)
        return true
    end
    local v = kc.Value
    if keypress and keyrelease and v >= 97 and v <= 122 then
        pcall(keypress, v - 32)
        task.wait(hold)
        pcall(keyrelease, v - 32)
        return true
    end
    return false
end

local function abilitySlot(kc)
    local name
    if kc == Enum.KeyCode.E then name = "RightAbility" elseif kc == Enum.KeyCode.Q then name = "LeftAbility" end
    if not name then return nil end
    return child(pgui(), "abilities", "Frame", name, "slot")
end

local function abilityReady(kc)
    local slot = abilitySlot(kc)
    if not slot then return true end
    local cd = slot:FindFirstChild("cooldownNumber")
    if cd and cd:IsA("TextLabel") and cd.Visible then
        local n = tonumber(cd.Text)
        if n and n > 0.05 then return false end
    end
    return true
end

function Input.ability(kc)
    if CFG.PressMode ~= "Button" then Input.key(kc) end
    if CFG.PressMode ~= "Key" then
        local slot = abilitySlot(kc)
        local b = slot and slot:FindFirstChildWhichIsA("GuiButton", true)
        if b then Input.fire(b) end
    end
end

local function slotIcon(kc)
    local slot = abilitySlot(kc)

    if b and b.Image ~= "" then return b.Image end

end

local function calibrate(kc)
    S.calibrating = true
    pcall(function()
        local _, hum = getChar()
        local icon = slotIcon(kc)
        if not hum or not icon then return end
        local before = hum.WalkSpeed
        Input.ability(kc)
        local peak = before
        local t0 = os.clock()
        while os.clock() - t0 < 1.2 do
            task.wait(0.1)
            local _, h = getChar()
            if h and h.WalkSpeed > peak then peak = h.WalkSpeed end
        end
        local isSpeed = peak >= before + 1.5
        S.speedIcons[icon] = isSpeed
        saveSpeed()
        log(string.format("calibrated %s: speed ability=%s (%.1f -> %.1f)", tostring(kc), tostring(isSpeed), before, peak))
    end)
    S.calibrating = false
end

function Scanner.list(pos)
    local out = {}
    local now = os.clock()
    for h, model in pairs(Scanner.hum) do
        if not h.Parent then
            Scanner.hum[h] = nil
        elseif h.Health > 0 and h.MaxHealth >= CFG.MinEnemyMaxHealth and not (S.black[h] and S.black[h] > now) and Scanner.isEnemy(model) then
            local root = model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
            if root then
                local d = ((root.Position - pos) * Vector3.new(1, 0, 1)).Magnitude
                local n = model.Name:lower()
                local score = d
                local boss = Scanner.isBoss(model)
                if boss then score = score - 80 end
                for k, v in pairs(CFG.PriorityNames) do
                    if n:find(k, 1, true) then score = score + v end
                end
                out[#out + 1] = { h = h, model = model, root = root, dist = d, score = score, boss = boss }
            end
        end
    end
    table.sort(out, function(a, b) return a.score < b.score end)
    return out
end

local function pickTarget(list)
    if #list == 0 then
        S.target = nil
        return nil
    end
    local best = list[1]
    if S.target then
        for _, e in ipairs(list) do
            if e.h == S.target and best.score > e.score - 25 then
                return e
            end
        end
    end
    S.target = best.h
    return best
end

local function zoneShape(p)
    local s, cf = p.Size, p.CFrame
    if p:IsA("Part") then
        if p.Shape == Enum.PartType.Ball then return "circle", s.X / 2 end
        if p.Shape == Enum.PartType.Cylinder then
            if math.abs(cf.RightVector.Y) > 0.8 then return "circle", math.max(s.Y, s.Z) / 2 end
            return nil
        end
    end
    local ax = { cf.RightVector, cf.UpVector, cf.LookVector }
    local vi, best = 1, -1
    for i = 1, 3 do
        local y = math.abs(ax[i].Y)
        if y > best then best = y vi = i end
    end
    if best < 0.7 then return nil end
    local others = {}
    for i = 1, 3 do
        if i ~= vi then others[#others + 1] = i end
    end
    return "box", others
end

local function prepZone(z)
    local cf, sz = z.cf, z.size
    local c = cf.Position
    z.cx = c.X
    z.cz = c.Z
    if z.kind == "circle" then
        z.r = z.a
        return true
    end
    local axes = { cf.RightVector, cf.UpVector, cf.LookVector }
    local sizes = { sz.X, sz.Y, sz.Z }
    local a1, a2 = axes[z.a[1]], axes[z.a[2]]
    local m1 = math.sqrt(a1.X * a1.X + a1.Z * a1.Z)
    local m2 = math.sqrt(a2.X * a2.X + a2.Z * a2.Z)
    if m1 < 0.1 or m2 < 0.1 then return false end
    z.ux = a1.X / m1
    z.uz = a1.Z / m1
    z.vx = a2.X / m2
    z.vz = a2.Z / m2
    z.hu = sizes[z.a[1]] / 2
    z.hv = sizes[z.a[2]] / 2
    return true
end

local function clearance(z, px, pz)
    local rx, rz = px - z.cx, pz - z.cz
    if z.kind == "circle" then
        return math.sqrt(rx * rx + rz * rz) - z.r
    end
    local du = math.abs(rx * z.ux + rz * z.uz) - z.hu
    local dv = math.abs(rx * z.vx + rz * z.vz) - z.hv
    if du > dv then return du end
    return dv
end

local TELE_DEFAULTS = { longline = 1.25, passivecricle = 1.47, beamtracker = 0.8, orbprecast = 1.0, safespot = 3.3 }
local function teleDefault(name)
    local n = name:lower()
    for k, v in pairs(TELE_DEFAULTS) do
        if n:find(k, 1, true) then return v end
    end
    return CFG.BossTele
end

function Dodger.eval(zs, px, pz, tArr)
    local m, mh = math.huge, math.huge
    for i = 1, #zs do
        local zn = zs[i]
        if zn.t1 > tArr then
            local c = clearance(zn, px, pz)
            local a = c - zn.pad
            if a < m then m = a end
            local b = c - CFG.ZoneHard
            if b < mh then mh = b end
        end
    end
    local sg = zs.safe
    if sg then
        for i = 1, #sg do
            local g = sg[i]
            if g.t1 > tArr then
                local best, bestR = -math.huge, 10
                for j = 1, #g.circles do
                    local c = g.circles[j]
                    local dx, dz = px - c.cx, pz - c.cz
                    local ins = c.r - math.sqrt(dx * dx + dz * dz)
                    if ins > best then
                        best = ins
                        bestR = c.r
                    end
                end
                local a = best - math.min(CFG.SafePref, bestR * 0.45)
                if a < m then m = a end
                local b = best - math.min(CFG.SafeHard, bestR * 0.25)
                if b < mh then mh = b end
            end
        end
    end
    return m, mh
end

function Dodger.unsafe(zs, pos, now)
    local m = Dodger.eval(zs, pos.X, pos.Z, now)
    return m < 0
end

function Dodger.has(zs)

end

end

local function onPartAdded(p)
    if not p.Anchored or p.CanCollide then return end
    local nm = p.Name:lower()
    if nm:find("orb", 1, true) and nm:find("boss", 1, true) and p:IsA("Part") and p.Shape == Enum.PartType.Ball then
        Dodger.zones[p] = {
            p = p, born = os.clock(), removed = nil, kind = "circle", a = p.Size.X / 2 + 6, cf = p.CFrame, size = p.Size,
            isPre = false, isHit = false, pre = false, hit = false, par = p.Name, boss = true, isOrb = true,
        }
        return
    end
    local isPre = nm == "precast" or nm == "circleprecast"
    local isHit = nm == "hitbox"
    if isPre or isHit then
        local kind, a = zoneShape(p)
        local par = p.Parent
        local pname = par and par.Name or "?"
        local lname = pname:lower()
        local boss = lname:find("boss", 1, true) ~= nil
        local g
        if boss and par then
            g = Dodger.groups[par]
            if not g then
                local t0 = teleDefault(pname)
                local memSafe = lname:find("memorysafezone", 1, true) ~= nil
                local memDmg = lname:find("memorydamagezone", 1, true) ~= nil
                g = {
                    model = par, name = pname, born = os.clock(), preSeen = false, preAlive = 0, preMax = 0,
                    firstGone = nil, preGone = nil, impact = os.clock() + t0, t0 = os.clock(), t1 = os.clock() + t0 + 1.5,
                    safe = (not memSafe) and (lname:find("safespot", 1, true) ~= nil or lname:find("safezone", 1, true) ~= nil),
                    orb = lname:find("orb", 1, true) ~= nil, tele0 = t0, memSafe = memSafe, memDmg = memDmg,
                }
                Dodger.groups[par] = g
                if memSafe then
                    Dodger.mem.lastEvent = os.clock()
                    log("memory circle appeared")
                end
            end
        end
        Dodger.zones[p] = {
            p = p, born = os.clock(), removed = nil, kind = kind, a = a, cf = p.CFrame, size = p.Size,
            isPre = isPre, isHit = isHit, pre = false, hit = false, par = pname, boss = boss, g = g,
        }
    end
end

local function onDesc(d)
    if d:IsA("BasePart") then
        onPartAdded(d)
    elseif d:IsA("Humanoid") then
        local model = d.Parent
        if not model or not model:IsA("Model") then return end
        if Players:GetPlayerFromCharacter(model) or model.Name == LP.Name then return end
        Scanner.hum[d] = model
    end
end

function Dodger.snapshot(now)
    local react = CFG.ReactionMs / 1000
    for _, g in pairs(Dodger.groups) do g.preAlive = 0 end
    for p, z in pairs(Dodger.zones) do
        local alive = p.Parent ~= nil
        if alive and z.isPre and p.Transparency > 0.97 then alive = false end
        if alive then
            local par = p.Parent
            z.pre = par:FindFirstChild("precast") ~= nil
            z.hit = par:FindFirstChild("hitBox") ~= nil
            local kind, a = zoneShape(p)
            if kind then
                if z.isOrb and a then a = a + 6 end
                z.kind = kind
                z.a = a
                z.cf = p.CFrame
                z.size = p.Size
            end
            z.removed = nil
            if z.g and z.isPre then
                z.g.preAlive = z.g.preAlive + 1
                z.g.preSeen = true
            end
        elseif not z.removed then
            z.removed = now
        end
    end
    for m, g in pairs(Dodger.groups) do
        local tele = Dodger.tele[g.name] or g.tele0
        if g.preAlive > g.preMax then g.preMax = g.preAlive end
        if g.preMax > 0 and g.preAlive < g.preMax and not g.firstGone then
            g.firstGone = now
            local d = now - g.born
            Dodger.tele[g.name] = tele * 0.6 + d * 0.4
            log(string.format("telegraph %s first strip gone after %.2fs", g.name, d))
        end
        if g.preSeen and g.preAlive == 0 and not g.preGone then
            g.preGone = now
        end
        local est = g.born + tele
        local impact
        if g.firstGone then
            impact = g.firstGone
        elseif g.preAlive > 0 and now > est then
            impact = now + 0.3
        else
            impact = est
        end
        g.impact = impact
        local endBase = g.preGone or (impact + 0.6)
        if g.safe then
            g.t1 = endBase + 1.0
            g.t0 = g.born
        elseif g.orb then
            g.t1 = endBase + 0.6
            g.t0 = g.born + react
        else
            g.t1 = endBase + CFG.Hit
            g.t0 = impact - CFG.Lead
        end
        if g.memDmg and now > g.t1 and not Dodger.mem.counted[g] then
            Dodger.mem.counted[g] = true
            if now - Dodger.mem.lastCount > 1.2 then
                Dodger.mem.lastCount = now
                Dodger.mem.attacks = Dodger.mem.attacks + 1
                Dodger.mem.lastEvent = now
                log("memory attack " .. Dodger.mem.attacks .. " resolved")
            end
        end
        if now > g.t1 + 0.5 or now - g.born > 9 then Dodger.groups[m] = nil end
    end
    local mem = Dodger.mem
    if next(mem.byGroup) and now - mem.lastEvent > 14 then
        mem.byGroup = {}
        mem.counted = {}
        mem.attacks = 0
    end
    local out = {}
    local safeMap = {}
    for p, z in pairs(Dodger.zones) do
        local keep = true
        local t0, t1
        if z.isOrb then
            if p.Parent == nil and (not z.removed or now - z.removed > 0.4) then keep = false end
            t0 = z.born
            t1 = now + 1
        elseif z.boss then
            local g = z.g
            if not g or Dodger.groups[g.model] ~= g then
                keep = false
            else
                t0 = math.max(g.t0, z.born + react)
                t1 = g.t1
            end
        else
            local endT = z.born + CFG.ZoneMaxLife
            if z.removed then
                endT = math.min(endT, math.max(z.born + CFG.ZoneMinLife, z.removed + CFG.Linger))
            end
            if now > endT then keep = false end
            t0 = z.born + react
            t1 = endT
        end
        if not keep then
            Dodger.zones[p] = nil
        elseif z.kind and now <= t1 then
            local use
            if z.isOrb then
                use = true
            elseif z.boss then
                local g = z.g
                if g.memDmg then
                    use = false
                elseif g.memSafe then
                    use = false
                    if z.isPre and z.kind == "circle" and prepZone(z) then
                        local e = mem.byGroup[g]
                        if not e then
                            e = { cx = z.cx, cz = z.cz, r = z.r, born = g.born }
                            mem.byGroup[g] = e
                        elseif z.r > e.r then
                            e.cx = z.cx
                            e.cz = z.cz
                            e.r = z.r
                        end
                    end
                elseif g.safe then
                    use = z.isPre and z.kind == "circle"
                elseif z.isHit then
                    use = true
                else
                    use = not z.hit
                end
            elseif z.isPre then
                use = not z.hit
            else
                use = z.pre
            end
            if use and prepZone(z) then
                z.t0 = t0
                z.t1 = t1
                z.pad = z.boss and CFG.BossPad or CFG.ZonePad
                if z.boss and z.g and z.g.safe then
                    local sg = safeMap[z.g]
                    if not sg then
                        sg = { t0 = t0, t1 = t1, circles = {}, name = z.g.name }
                        safeMap[z.g] = sg
                    end
                    sg.circles[#sg.circles + 1] = z
                else
                    out[#out + 1] = z
                end
            end
        end
    end
    out.safe = {}
    for _, sg in pairs(safeMap) do out.safe[#out.safe + 1] = sg end
    if next(mem.byGroup) then
        local sorted = {}
        for _, e in pairs(mem.byGroup) do sorted[#sorted + 1] = e end
        table.sort(sorted, function(a, b) return a.born < b.born end)
        local k = mem.attacks + 1
        if k <= #sorted then
            local e = sorted[k]
            out.safe[#out.safe + 1] = {
                t0 = now, t1 = now + 3, name = "memory",
                circles = { { cx = e.cx, cz = e.cz, r = e.r, t0 = now, t1 = now + 3, pad = 0 } },
            }
        end
    end
    return out
end

local function rayParams()
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = { LP.Character }
    pcall(function() rp.RespectCanCollide = true end)
    return rp
end

local function walkable(from, to, rp)
    local g = workspace:Raycast(to + Vector3.new(0, 6, 0), Vector3.new(0, -30, 0), rp)
    if not g then return nil end
    if workspace:Raycast(from + Vector3.new(0, 2, 0), to - from, rp) then return nil end
    return Vector3.new(to.X, g.Position.Y + 3, to.Z)
end

local function crossOk(from, to, zs, now, speed)
    local dx, dz = to.X - from.X, to.Z - from.Z
    local d = math.sqrt(dx * dx + dz * dz)
    local steps = math.max(1, math.floor(d / 3))
    for st = 1, steps do
        local f = st / steps
        local x, z = from.X + dx * f, from.Z + dz * f
        local t = now + (d * f) / speed
        for i = 1, #zs do
            local zn = zs[i]
            if t >= zn.t0 - 0.05 and t <= zn.t1 then
                if clearance(zn, x, z) < 0.3 then return false end
            end
        end
    end
    return true
end

function Dodger.escape(pos, speed, zs, now)
    local bestV, bestPt
    for i = 0, 15 do
        local a = (i / 16) * math.pi * 2
        local x, z = pos.X + math.cos(a) * 12, pos.Z + math.sin(a) * 12
        local _, mh = Dodger.eval(zs, x, z, now + 12 / (speed * CFG.SpeedMargin))
        if not bestV or mh > bestV then
            bestV = mh
            bestPt = Vector3.new(x, pos.Y, z)
        end
    end
    return bestPt
end

function Dodger.plan(pos, speed, desired, anchor, orbitR, zs, now)
    local rp = rayParams()
    local px, pz = pos.X, pos.Z
    local sp = speed * CFG.SpeedMargin
    local cands = {}
    local all = {}
    local best
    local bad = S.dBad
    local function nearBad(x, z)
        for i = 1, #bad do
            local b = bad[i]
            if b.t > now then
                local dx, dz = x - b.x, z - b.z
                if dx * dx + dz * dz < 36 then return true end
            end
        end
        return false
    end
    local function consider(x, z, d, far, keepAll)
        local m, mh = Dodger.eval(zs, x, z, now + d / sp)
        if not best or mh > best.mh then best = { x = x, z = z, far = far, mh = mh } end
        if keepAll then all[#all + 1] = { x = x, z = z, mh = mh } end
        if mh >= 0 and not nearBad(x, z) then
            local cost = d
            if m < 0 then cost = cost - m * 5 end
            if desired then
                local dx, dz = x - desired.X, z - desired.Z
                cost = cost + math.sqrt(dx * dx + dz * dz) * 0.3
            elseif anchor then
                local ax, az = x - anchor.X, z - anchor.Z
                cost = cost + math.abs(math.sqrt(ax * ax + az * az) - orbitR) * 0.25
            end
            cands[#cands + 1] = { x = x, z = z, cost = cost, far = far }
        end
    end
    local function ring(R, stepSize, keepAll)
        for ix = -R, R, stepSize do
            for iz = -R, R, stepSize do
                local d = math.sqrt(ix * ix + iz * iz)
                if d <= R then consider(px + ix, pz + iz, d, false, keepAll) end
            end
        end
    end
    local function climb(x, z)
        local stepLen = 1.5
        local d0 = math.sqrt((x - px) ^ 2 + (z - pz) ^ 2)
        local _, cur = Dodger.eval(zs, x, z, now + d0 / sp)
        for _ = 1, 6 do
            local bx, bz, bv = x, z, cur
            for k = 0, 7 do
                local a = (k / 8) * math.pi * 2
                local nx, nz = x + math.cos(a) * stepLen, z + math.sin(a) * stepLen
                local d = math.sqrt((nx - px) ^ 2 + (nz - pz) ^ 2)
                local _, v = Dodger.eval(zs, nx, nz, now + d / sp)
                if v > bv then bx, bz, bv = nx, nz, v end
            end
            if bx == x and bz == z then
                stepLen = stepLen / 2
            else
                x, z, cur = bx, bz, bv
            end
            if stepLen < 0.3 then break end
        end
        consider(x, z, math.sqrt((x - px) ^ 2 + (z - pz) ^ 2), false, false)
    end
    ring(30, 2.5, true)
    if #cands < 6 then
        table.sort(all, function(a, b) return a.mh > b.mh end)
        for i = 1, math.min(#all, 10) do climb(all[i].x, all[i].z) end
    end
    if #cands == 0 then
        ring(60, 5, false)
        if #cands == 0 and best then climb(best.x, best.z) end
    end
    if zs.safe then
        for _, sg in ipairs(zs.safe) do
            if sg.t1 > now then
                for _, c in ipairs(sg.circles) do
                    local dx, dz = c.cx - px, c.cz - pz
                    local dist = math.sqrt(dx * dx + dz * dz)
                    consider(c.cx, c.cz, dist, dist > 30, false)
                    if dist > 1 then
                        local inner = math.max(0, dist - (c.r - 8))
                        consider(px + dx / dist * inner, pz + dz / dist * inner, inner, inner > 30, false)
                    end
                end
            end
        end
    end
    table.sort(cands, function(a, b) return a.cost < b.cost end)
    local function resolve(c)
        if c.far then
            local g = workspace:Raycast(Vector3.new(c.x, pos.Y + 6, c.z), Vector3.new(0, -40, 0), rp)
            if g then return Vector3.new(c.x, g.Position.Y + 3, c.z) end
            return nil
        end
        return walkable(pos, Vector3.new(c.x, pos.Y, c.z), rp)
    end
    for i = 1, math.min(#cands, 16) do
        local c = cands[i]
        local w = resolve(c)
        if w and crossOk(pos, w, zs, now, sp) then return w end
    end
    for i = 1, math.min(#cands, 16) do
        local w = resolve(cands[i])
        if w then return w end
    end
    if best then
        local w = resolve(best)
        if w then return w end
        return Vector3.new(best.x, pos.Y, best.z)
    end
    return nil
end

function Dodger.goal(pos, speed, desired, anchor, orbitR, zs, now)
    local g = S.dGoal
    if g and now - S.dGoalT < 2.5 then
        local dx, dz = g.X - pos.X, g.Z - pos.Z
        local d = math.sqrt(dx * dx + dz * dz)
        if d > 2.5 then
            local _, mh = Dodger.eval(zs, g.X, g.Z, now + d / (speed * CFG.SpeedMargin))
            if mh >= 0 then return g end
        end
    end
    local w = Dodger.plan(pos, speed, desired, anchor, orbitR, zs, now)
    if not w then w = Dodger.escape(pos, speed, zs, now) end
    if w then
        S.dGoal = w
        S.dGoalT = now
    end
    return w
end

local Aim = { att = nil, ao = nil, dir = nil, untilT = 0, lockUntil = 0, locked = false }

function Aim.ensure(hrp)
    if Aim.ao and Aim.ao.Parent == hrp and Aim.att and Aim.att.Parent == hrp then return true end
    if Aim.ao then pcall(function() Aim.ao:Destroy() end) end
    if Aim.att then pcall(function() Aim.att:Destroy() end) end
    Aim.ao = nil
    Aim.att = nil
    local ok = pcall(function()
        local att = Instance.new("Attachment")
        att.Name = "ExodusAimAttachment"
        att.Parent = hrp
        local ao = Instance.new("AlignOrientation")
        ao.Name = "ExodusAim"
        ao.Mode = Enum.OrientationAlignmentMode.OneAttachment
        ao.Attachment0 = att
        ao.RigidityEnabled = false
        ao.MaxTorque = 1e9
        ao.MaxAngularVelocity = 100
        ao.Responsiveness = 60
        ao.Enabled = false
        ao.Parent = hrp
        Aim.att = att
        Aim.ao = ao
    end)
    return ok and Aim.ao ~= nil
end

function Aim.face(dir)
    if CFG.FaceMode == "Off" then return end
    local v = Vector3.new(dir.X, 0, dir.Z)
    if v.Magnitude < 0.01 then return end
    Aim.dir = v.Unit
    Aim.untilT = os.clock() + 0.25
    Aim.lockUntil = math.max(Aim.lockUntil, os.clock() + 0.3)
end

function Aim.snap(tpos)
    local cam = workspace.CurrentCamera
    if CFG.CamAim and cam then
        local cp = cam.CFrame.Position
        if (tpos - cp).Magnitude > 1 then cam.CFrame = CFrame.lookAt(cp, tpos) end
    end
    if CFG.FaceMode == "Off" then return end
    local _, _, hrp = getChar()
    if not hrp then return end
    local pos = hrp.Position
    local v = Vector3.new(tpos.X - pos.X, 0, tpos.Z - pos.Z)
    if v.Magnitude < 0.1 then return end
    Aim.face(v)
    hrp.CFrame = CFrame.lookAt(pos, pos + v)
end

function Aim.release()
    Aim.dir = nil
    Aim.lockUntil = 0
    if Aim.ao then pcall(function() Aim.ao.Enabled = false end) end
    local _, hum = getChar()
    if hum and Aim.locked then hum.AutoRotate = true end
    Aim.locked = false
end

function Aim.update()
    local _, hum, hrp = getChar()
    if not hrp then return end
    local now = os.clock()
    local want = now < Aim.lockUntil and CFG.FaceMode ~= "Off"
    if want and not Aim.locked then
        hum.AutoRotate = false
        Aim.locked = true
    elseif not want and Aim.locked then
        hum.AutoRotate = true
        Aim.locked = false
    end
    local active = Aim.dir ~= nil and now < Aim.untilT and CFG.FaceMode ~= "Off"
    if active and CFG.FaceMode == "Constraint" then
        if Aim.ensure(hrp) then
            Aim.ao.CFrame = CFrame.lookAt(Vector3.new(0, 0, 0), Aim.dir)
            Aim.ao.Enabled = true
        end
    else
        if Aim.ao and Aim.ao.Enabled then Aim.ao.Enabled = false end
        if active and CFG.FaceMode == "CFrame" then
            local pos = hrp.Position
            hrp.CFrame = CFrame.lookAt(pos, pos + Aim.dir)
        end
    end
end

local Cam = { last = 0 }

    if not CFG.LookCam then return end

    local cam = workspace.CurrentCamera
    if not cam then return end
    local cp = cam.CFrame.Position

    cam.CFrame = CFrame.lookAt(cp, pos)
end

local Mover = {
    goal = nil, want = nil, alts = nil, wps = nil, wi = 2, dirty = false, busy = false, lastPos = nil, lastT = 0,
    stuck = 0, lastCompute = 0, sideUntil = 0, sidePt = nil, failed = false, failAt = 0, flip = false,
}

local AGENT_RADII = { 2.5, 1.4, 0.7 }

function Mover.clear()
    Mover.goal = nil
    Mover.want = nil
    Mover.alts = nil
    Mover.wps = nil
    Mover.dirty = false
    Mover.stuck = 0
    Mover.sidePt = nil
    Mover.failed = false
end

function Mover.setGoal(g, alts)
    if alts then Mover.alts = alts end
    if Mover.want and (Mover.want - g).Magnitude < 5 then return end
    local now = os.clock()
    if Mover.want and Mover.wps and now - Mover.lastCompute < 0.7 then
        Mover.want = g
        return
    end
    Mover.want = g
    Mover.goal = g
    Mover.wps = nil
    Mover.dirty = true
end

local function computePath(from)
    if Mover.busy then return end
    Mover.busy = true
    Mover.dirty = false
    Mover.lastCompute = os.clock()
    local want = Mover.want
    local alts = Mover.alts
    task.spawn(function()
        local chosen, wps
        local tries = { want }
        if alts then
            for i = 1, math.min(#alts, 3) do tries[#tries + 1] = alts[i] end
        end
        for ti, g in ipairs(tries) do
            for ri, rad in ipairs(AGENT_RADII) do
                if not (ti > 1 and ri > 2) then
                    local path = PathfindingService:CreatePath({ AgentRadius = rad, AgentHeight = 5, AgentCanJump = true, WaypointSpacing = 6 })
                    local ok = pcall(function() path:ComputeAsync(from, g) end)
                    if ok and path.Status == Enum.PathStatus.Success then
                        chosen = g
                        wps = path:GetWaypoints()
                        break
                    end
                end
            end
            if wps or Mover.want ~= want then break end
        end
        if Mover.want == want then
            if wps then
                Mover.goal = chosen
                Mover.wps = wps
                Mover.wi = 2
                Mover.failed = false
            else
                Mover.goal = want
                Mover.wps = {}
                Mover.failed = true
                Mover.failAt = os.clock()
            end
        end
        Mover.busy = false
    end)
end

local function sidestepPoint(pos, goal, flip)
    local flat = Vector3.new(1, 0, 1)
    local fwd = (goal - pos) * flat
    if fwd.Magnitude < 0.1 then fwd = Vector3.new(1, 0, 0) end
    fwd = fwd.Unit
    local left = Vector3.new(-fwd.Z, 0, fwd.X)
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = { LP.Character }
    pcall(function() rp.RespectCanCollide = true end)
    local dirs = { left, -left, (left + fwd).Unit, (-left + fwd).Unit, (left - fwd).Unit, (-left - fwd).Unit }
    if flip then dirs = { -left, left, (-left + fwd).Unit, (left + fwd).Unit, (-left - fwd).Unit, (left - fwd).Unit } end
    local best, bestD
    for _, d in ipairs(dirs) do
        local r = workspace:Raycast(pos + Vector3.new(0, 1.5, 0), d * 12, rp)
        local free = r and (r.Position - pos).Magnitude or 12
        if not bestD or free > bestD + 1.5 then
            bestD = free
            best = d
        end
    end
    return pos + best * math.max(4, math.min(bestD - 1.5, 10))
end

function Mover.step(hum, hrp)
    local goal = Mover.goal
    if not goal then return nil end
    local pos = hrp.Position
    local now = os.clock()
    local flatV = Vector3.new(1, 0, 1)
    if Mover.sidePt and now < Mover.sideUntil then
        hum:MoveTo(Mover.sidePt)
        return "moving"
    end
    if ((goal - pos) * flatV).Magnitude < 3.5 then
        Mover.clear()
        return "arrived"
    end
    if Mover.dirty then computePath(pos) end
    local wps = Mover.wps
    local wp = wps and wps[Mover.wi]
    while wp and ((wp.Position - pos) * flatV).Magnitude < 3.5 do
        Mover.wi = Mover.wi + 1
        wp = wps[Mover.wi]
    end
    if wp then
        if wp.Action == Enum.PathWaypointAction.Jump then hum.Jump = true end
        hum:MoveTo(wp.Position)
    else
        hum:MoveTo(goal)
    end
    if now - Mover.lastT >= 0.8 then
        if Mover.lastPos and (pos - Mover.lastPos).Magnitude < 1.0 then
            Mover.stuck = Mover.stuck + 1
            hum.Jump = true
            if Mover.stuck == 2 or Mover.stuck == 4 then
                Mover.flip = not Mover.flip
                Mover.sidePt = sidestepPoint(pos, goal, Mover.flip)
                Mover.sideUntil = now + 0.9
                log("stuck: sidestep")
            end
            if Mover.stuck >= 3 then
                Mover.wps = nil
                Mover.dirty = true
            end
            if Mover.stuck >= 7 then
                Mover.stuck = 0
                Mover.lastPos = pos
                Mover.lastT = now
                return "stuck"
            end
        else
            Mover.stuck = 0
        end
        Mover.lastPos = pos
        Mover.lastT = now
    end
    return "moving"
end

local function isStunned()
    local sp = workspace:FindFirstChild("stunParts")
    return sp ~= nil and sp:FindFirstChild(LP.Name) ~= nil
end

local Combat = {}

local function castWithAim(kc, tpos)
    pcall(function()
        if CFG.FaceMode ~= "Off" or CFG.CamAim then Aim.snap(tpos) end
        RunService.Heartbeat:Wait()
        Input.ability(kc)
    end)
    S.casting = false
end

function Combat.cast(now, hum, hrp, tpos)
    if S.casting or S.calibrating or now < S.nextCast or now < S.castHold then return false end
    if hum.WalkSpeed < 4 or hrp.Anchored or hum.PlatformStand or isStunned() then return false end
    local keys = {}
    if CFG.UseFirst then keys[#keys + 1] = CFG.FirstKey end
    if CFG.UseSecond then keys[#keys + 1] = CFG.SecondKey end
    for _, k in ipairs(keys) do
        local sp = isSpeedSlot(k)
        local skip = CFG.AlwaysSpeed and (sp == true or (sp == nil and CFG.SpeedSlot == "Auto" and slotIcon(k) ~= nil))
        if not skip and abilityReady(k) then
            S.casting = true
            S.nextCast = now + CFG.CastGap * CFG.Humanize * jitter(0.85, 1.3)
            S.lastCast = now
            task.spawn(castWithAim, k, tpos)
            return true
        end
    end
    return false
end

function Combat.speedTick(now, hum, hrp, unsafe)
    if not CFG.AlwaysSpeed or S.casting or S.calibrating or unsafe then return false end
    if now < S.nextSpeed then return false end
    if hum.WalkSpeed < 4 or hrp.Anchored or hum.PlatformStand or isStunned() then return false end
    if not S.speedScanned then scanSpeedIcons() end
    for _, kc in ipairs({ CFG.FirstKey, CFG.SecondKey }) do
        local sp = isSpeedSlot(kc)
        if sp == true and abilityReady(kc) then
            S.nextSpeed = now + 0.35
            task.spawn(Input.ability, kc)
            return true
        elseif sp == nil and CFG.SpeedSlot == "Auto" and abilityReady(kc) and now - S.lastCalib > 1.0 and slotIcon(kc) then
            S.lastCalib = now
            task.spawn(calibrate, kc)
            return true
        end
    end
    return false
end

local Hide = { saved = setmetatable({}, { __mode = "k" }), running = false }

local function hideApply(d)
    if d:IsA("BasePart") then
        d.LocalTransparencyModifier = 1
    elseif d:IsA("Decal") or d:IsA("Texture") then
        if Hide.saved[d] == nil then Hide.saved[d] = d.Transparency end
        d.Transparency = 1
    elseif d:IsA("BillboardGui") or d:IsA("ParticleEmitter") or d:IsA("Trail") or d:IsA("Beam") or d:IsA("Fire") or d:IsA("Smoke") or d:IsA("Sparkles") then
        if Hide.saved[d] == nil then Hide.saved[d] = d.Enabled end
        d.Enabled = false
    elseif d:IsA("Humanoid") then
        d.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
    end
end

local function hideRestore()
    for d, v in pairs(Hide.saved) do
        pcall(function()
            if d:IsA("Decal") or d:IsA("Texture") then d.Transparency = v else d.Enabled = v end
        end)
    end
    Hide.saved = setmetatable({}, { __mode = "k" })
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= LP and pl.Character then
            for _, d in ipairs(pl.Character:GetDescendants()) do
                if d:IsA("BasePart") then pcall(function() d.LocalTransparencyModifier = 0 end) end
                if d:IsA("Humanoid") then pcall(function() d.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.Viewer end) end
            end
        end
    end
end

function Hide.set(on)
    if on then
        if Hide.running then return end
        Hide.running = true
        task.spawn(function()
            while CFG.HidePlayers and alive do
                for _, pl in ipairs(Players:GetPlayers()) do
                    if pl ~= LP and pl.Character then
                        for _, d in ipairs(pl.Character:GetDescendants()) do pcall(hideApply, d) end
                    end
                end
                task.wait(1.5)
            end
            Hide.running = false
        end)
    else
        hideRestore()
    end
end

local function lookAwayActive(now)
    if now < S.laCheck then return S.lookAway end
    S.laCheck = now + 0.1
    S.lookAway = Gui.find({ "look away" }, false) ~= nil
    return S.lookAway
end

local function roomsInfo()
    local out = {}
    local d = workspace:FindFirstChild("dungeon")
    if not d then return out end
    for _, c in ipairs(d:GetChildren()) do
        local ef = c:FindFirstChild("enemyFolder")
        if ef then
            local sum, n = Vector3.new(0, 0, 0), 0
            for _, sp in ipairs(ef:GetChildren()) do
                if sp:IsA("BasePart") then
                    sum = sum + sp.Position
                    n = n + 1
                end
            end
            if n > 0 then
                local ov = c:FindFirstChild("order")
                local order = tonumber(c.Name:match("%d+")) or 999
                if ov and ov:IsA("IntValue") then order = ov.Value end
                if c.Name == "bossRoom" then order = 1000 end
                local spawns = {}
                for _, sp in ipairs(ef:GetChildren()) do
                    if sp:IsA("BasePart") then spawns[#spawns + 1] = sp.Position end
                end
                out[#out + 1] = { name = c.Name, order = order, pos = sum / n, spawns = spawns, barrier = c:FindFirstChild("barrier") ~= nil }
            end
        end
    end
    table.sort(out, function(a, b) return a.order < b.order end)
    return out
end

local function roomsPending()
    for _, r in ipairs(roomsInfo()) do
        if not S.roomDone[r.name] then return true end
    end
    return false
end

local function nextRoomGoal(pos, now)
    local rooms = roomsInfo()
    if #rooms == 0 then return nil, false end
    local flat = Vector3.new(1, 0, 1)
    for _, r in ipairs(rooms) do
        if not S.roomDone[r.name] then
            S.room = r.order
            S.roomSince[r.name] = S.roomSince[r.name] or now
            local d = ((r.pos - pos) * flat).Magnitude
            if now - S.roomSince[r.name] > 60 then
                S.roomDone[r.name] = true
                log("room skipped (timeout): " .. r.name)
            elseif d < 30 then
                S.roomIdle[r.name] = S.roomIdle[r.name] or now
                if now - S.roomIdle[r.name] > 4 then
                    S.roomDone[r.name] = true
                    log("room done: " .. r.name)
                end
                return nil, true
            else
                if r.barrier and Mover.failed and now - Mover.failAt < 3 then
                    return nil, true
                end
                local alts = {}
                for _, sp in ipairs(r.spawns) do alts[#alts + 1] = sp end
                table.sort(alts, function(a, b) return (a - pos).Magnitude < (b - pos).Magnitude end)
                return r.pos, true, alts
            end
        end
    end
    return nil, true
end

local function watchProgress(now, pos, target, hum)
    if not S.sampleT or now - S.sampleT > 3 then
        S.sampleT = now
        S.samplePos = pos
        S.ticks = 0
        S.moveTicks = 0
        return
    end
    S.ticks = S.ticks + 1
    if S.wantMove then S.moveTicks = S.moveTicks + 1 end
    if now - S.sampleT < 0.9 then return end
    local moved = ((pos - S.samplePos) * Vector3.new(1, 0, 1)).Magnitude
    local wanted = S.ticks > 0 and (S.moveTicks / S.ticks) > 0.7
    S.sampleT = now
    S.samplePos = pos
    S.ticks = 0
    S.moveTicks = 0
    if wanted and moved < 4 then
        S.castHold = now + 0.9
        S.blockHits = S.blockHits + 1
        log(string.format("blocked: moved %.1f hits %d", moved, S.blockHits))
        if now < S.lookUntil then
            S.aimOff = now + 3
            Aim.release()
        end
        if S.blockHits >= 5 then
            S.black[target.h] = now + 10
            S.blockHits = 0
            Mover.clear()
            hum.Jump = true
            log("target blacklisted: unreachable")
        end
    elseif moved >= 8 then
        S.blockHits = 0
    end
end

local function ringAlts(tpos, r, pos)
    local out = {}
    for i = 0, 7 do
        local a = (i / 8) * math.pi * 2
        out[#out + 1] = Vector3.new(tpos.X + math.cos(a) * r, pos.Y, tpos.Z + math.sin(a) * r)
    end
    table.sort(out, function(x, y) return (x - pos).Magnitude < (y - pos).Magnitude end)
    return out
end

local function orbitGoal(pos, tpos, r)
    local rel = (pos - tpos) * Vector3.new(1, 0, 1)
    if rel.Magnitude < 1 then rel = Vector3.new(1, 0, 0) end
    local ang = math.atan2(rel.Z, rel.X) + S.orbitDir * 0.5
    return Vector3.new(tpos.X + math.cos(ang) * r, pos.Y, tpos.Z + math.sin(ang) * r)
end

local function goTo(hum, hrp, goal, pos)
    local d = ((goal - pos) * Vector3.new(1, 0, 1)).Magnitude
    if d > 28 then
        Mover.setGoal(goal)
        Mover.step(hum, hrp)
    else
        Mover.clear()
        hum:MoveTo(goal)
    end
end

local function dodgeWatch(now, pos, hum)
    local flat = Vector3.new(1, 0, 1)
    if not S.dPos or now - S.dT > 1.5 then
        S.dPos = pos
        S.dT = now
        return
    end
    if now - S.dT >= 0.5 then
        local moved = ((pos - S.dPos) * flat).Magnitude
        S.dPos = pos
        S.dT = now
        if moved < 0.8 and not isStunned() then
            hum.Jump = true
            if S.dGoal then
                S.dBad[#S.dBad + 1] = { x = S.dGoal.X, z = S.dGoal.Z, t = now + 3 }
                if #S.dBad > 8 then table.remove(S.dBad, 1) end
            end
            S.dGoal = nil
            Mover.clear()
            if now < S.lookUntil then
                S.aimOff = now + 3
                Aim.release()
            end
            log("dodge stuck, replanning")
        end
    end
end

local function brain(hum, hrp)
    local now = os.clock()
    local pos = hrp.Position
    local flat = Vector3.new(1, 0, 1)
    local list = Scanner.list(pos)
    if #list > 0 then
        S.lastEnemyT = now
        S.roomIdle = {}
    end
    local target = pickTarget(list)
    local boss
    for _, e in ipairs(list) do
        if e.boss then boss = e break end
    end
    if boss then S.bossPos = boss.root.Position end

    local lookAway = false
    if CFG.FaceAway then
        if lookAwayActive(now) then
            if now >= S.lookUntil then log("look away detected") end
            S.lookUntil = now + CFG.LookHold
        end
        if now < S.lookUntil and (boss or S.bossPos) then lookAway = true end
    end
    if lookAway then
        local bp = boss and boss.root.Position or S.bossPos
        local away = (pos - bp) * flat
        if away.Magnitude < 0.1 then away = Vector3.new(1, 0, 0) end
        if now >= S.aimOff then Aim.face(away) end
        Cam.snap(pos + away.Unit * 60)
    end

    local zs = {}
    if CFG.DodgeZones then zs = Dodger.snapshot(now) end
    drawDebug(zs, now)
    local speed = math.max(hum.WalkSpeed, 8)
    local has = Dodger.has(zs)
    local unsafe = has and Dodger.unsafe(zs, pos, now)
    local holdSafe = has and not unsafe and Dodger.safePending(zs, now)
    if not has then S.dGoal = nil end
    Combat.speedTick(now, hum, hrp, unsafe)

    if target then
        local tpos = target.root.Position
        local orbitR = target.boss and CFG.BossOrbit or CFG.OrbitRadius
        if CFG.ContAim and not lookAway then Aim.face(tpos - pos) end
        local dist = ((tpos - pos) * flat).Magnitude
        local near = dist <= orbitR + 14
        local moving = false
        local dodging = false
        local holding = false
        if unsafe then
            dodging = true
            local goal = Dodger.goal(pos, speed, nil, tpos, orbitR, zs, now)
            if goal then
                goTo(hum, hrp, goal, pos)
                moving = true
            end
            dodgeWatch(now, pos, hum)
        elseif holdSafe then
            Mover.clear()
            hum:MoveTo(pos)
            holding = true
        elseif near then
            Mover.clear()
            if CFG.Orbit then
                local desired = orbitGoal(pos, tpos, orbitR)
                local safe = true
                if has then
                    local dd = ((desired - pos) * flat).Magnitude
                    local m = Dodger.eval(zs, desired.X, desired.Z, now + dd / speed)
                    safe = m >= 0
                end
                if safe then
                    local w = walkable(pos, desired, rayParams())
                    if w then
                        hum:MoveTo(w)
                        moving = true
                    else
                        S.orbitDir = -S.orbitDir
                        local rel = (pos - tpos) * flat
                        if rel.Magnitude < 1 then rel = Vector3.new(1, 0, 0) end
                        Mover.setGoal(tpos + rel.Unit * orbitR, ringAlts(tpos, orbitR, pos))
                        moving = Mover.step(hum, hrp) ~= nil
                    end
                else
                    local goal = Dodger.goal(pos, speed, desired, tpos, orbitR, zs, now)
                    if goal then
                        goTo(hum, hrp, goal, pos)
                        moving = true
                    else
                        hum:MoveTo(pos)
                        holding = true
                    end
                end
            else
                hum:MoveTo(pos)
            end
        else
            local rel = (pos - tpos) * flat
            if rel.Magnitude < 1 then rel = Vector3.new(1, 0, 0) end
            Mover.setGoal(tpos + rel.Unit * orbitR, ringAlts(tpos, orbitR, pos))
            moving = true
            if Mover.step(hum, hrp) == "stuck" then
                S.black[target.h] = now + 8
                Mover.clear()
                log("target blacklisted: path stuck")
            end
        end
        S.wantMove = moving
        if lookAway then
            S.state = dodging and "LOOK AWAY + DODGE" or "LOOK AWAY"
        elseif dodging then
            S.state = "DODGE"
        elseif holdSafe then
            S.state = "SAFE CIRCLE"
            if near then Combat.cast(now, hum, hrp, tpos) end
        elseif near then
            S.state = (target.boss and "BOSS " or "") .. (holding and "HOLD" or "ORBIT")
            Combat.cast(now, hum, hrp, tpos)
        else
            S.state = target.boss and "BOSS APPROACH" or "APPROACH"
        end
        watchProgress(now, pos, target, hum)
        return
    end

    S.wantMove = false

    if unsafe then
        local goal = Dodger.goal(pos, speed, nil, nil, CFG.OrbitRadius, zs, now)
        dodgeWatch(now, pos, hum)
        if goal then
            S.state = lookAway and "LOOK AWAY + DODGE" or "DODGE"
            goTo(hum, hrp, goal, pos)
            return
        end
    end
    if holdSafe then
        S.state = lookAway and "LOOK AWAY" or "SAFE CIRCLE"
        Mover.clear()
        hum:MoveTo(pos)
        return
    end

    local rg, haveRooms, ralts = nil, false, nil
    if CFG.AutoNav then rg, haveRooms, ralts = nextRoomGoal(pos, now) end
    if rg then
        S.state = lookAway and "LOOK AWAY" or ("NEXT ROOM " .. tostring(S.room))
        Mover.setGoal(rg, ralts)
        if Mover.step(hum, hrp) == "stuck" then
            S.roomStuck = (S.roomStuck or 0) + 1
            Mover.clear()
            if S.roomStuck >= 3 then
                S.roomStuck = 0
                for _, r in ipairs(roomsInfo()) do
                    if not S.roomDone[r.name] then
                        S.roomDone[r.name] = true
                        log("room skipped (stuck): " .. r.name)
                        break
                    end
                end
            end
        end
        return
    end
    if haveRooms then
        S.state = lookAway and "LOOK AWAY" or "WAITING IN ROOM"
        Mover.clear()
        return
    end

    S.state = "WAITING"
    Mover.clear()
end

local function inDungeon(pos)
    if game.PlaceId == CFG.DungeonPlace then return true end
    if game.PlaceId == CFG.LobbyPlace then return false end
    return #Scanner.list(pos) > 0
end

local function tryFinish(leave)
    if Gui.find(CFG.EndTexts, false) then
        for _, b in ipairs(CFG.EndButtons) do
            local g = Gui.find({ b }, true)
            if g then
                log("finish click: " .. b)
                press(toButton(g), true)
                return true
            end
        end
    end
    local conf = Gui.find({ "yes", "confirm" }, true)
    if conf and os.clock() >= S.nextConfirm then
        S.nextConfirm = os.clock() + 3
        log("confirm click")
        press(toButton(conf), true)
        return true
    end
    if leave then
        local btn = child(pgui(), "mainInterface", "buttons", "optionsButton", "ReturnToLobbyButton")
        if btn then
            log("return to lobby button")
            press(btn, false)
            return true
        end
    end
    return false
end

local function promptPos(pr)
    local par = pr.Parent
    if not par then return nil end
    if par:IsA("BasePart") then return par.Position end
    if par:IsA("Attachment") then return par.WorldPosition end
    if par:IsA("Model") then return par:GetPivot().Position end
    return nil
end

local function findPrompt(words)
    for _, d in ipairs(workspace:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Enabled then
            local par = d.Parent
            local hay = (d.ObjectText .. " " .. d.ActionText .. " " .. (par and par.Name or "") .. " " .. ((par and par.Parent) and par.Parent.Name or "")):lower()
            for _, w in ipairs(words) do
                if hay:find(w, 1, true) then return d end
            end
        end
    end
    return nil
end

local function triggerPrompt(pr)
    if fireproximityprompt then
        pcall(fireproximityprompt, pr)
    else
        Input.key(pr.KeyboardKeyCode, pr.HoldDuration + 0.25)
    end
end

local function dungeonStartStep(hum, hrp, now)
    if now >= S.nextStartTry then
        S.nextStartTry = now + jitter(1.8, 2.8)
        local sb = child(pgui(), "queueGui", "lobbyInfo", "startBackground", "startFrame", "startButton")
        if sb and isShown(sb) then
            log("dungeon start: party start button")
            press(sb, false)
            return
        end
        local g = Gui.find({ "start", "start dungeon", "start game", "begin", "ready" }, true)
        if g then
            log("dungeon start: " .. trimText(g))
            press(toButton(g), false)
            return
        end
    end
    local waited = now - (S.waitT or now)
    if waited > 4 then
        if now >= S.nextPromptScan then
            S.nextPromptScan = now + 8
            S.startPrompt = findPrompt({ "start", "begin", "ready" })
        end
        local pr = S.startPrompt
        if pr and pr.Parent then
            local pp = promptPos(pr)
            if pp then
                local d = ((pp - hrp.Position) * Vector3.new(1, 0, 1)).Magnitude
                if d > 8 then
                    Mover.setGoal(pp)
                    Mover.step(hum, hrp)
                else
                    Mover.clear()
                    if now >= S.nextPrompt then
                        S.nextPrompt = now + 4
                        log("dungeon start: prompt")
                        triggerPrompt(pr)
                    end
                end
                return
            end
        end
    end
    if waited > 15 and now >= S.nextHost then
        S.nextHost = now + 20
        local hb = child(pgui(), "mainInterface", "buttons", "optionsButton", "HostButton")
        if hb and isShown(hb) then
            log("dungeon start: host button")
            press(hb, false)
        end
    end
end

local function lobbyFlow(now)
    local qg = child(pgui(), "queueGui")
    local info = child(qg, "lobbyInfo")
    local cd = child(qg, "chooseDungeon")
    local sel = child(qg, "selectOption")
    local function go(key, btn, idem, gap)
        if not btn then return false end
        if S.clicked[key] and now - S.clicked[key] < (gap or 4) then return true end
        S.clicked[key] = now
        log("lobby: " .. key)
        press(btn, idem)
        return true
    end
    if info and isShown(info) then
        return go("lobbyStart", child(info, "startBackground", "startFrame", "startButton"), false, 6)
    end
    if cd and isShown(cd) then
        local mid = child(cd, "backgroundFillMiddle", "dungeonName")
        local picked = mid and mid:IsA("TextLabel") and trimText(mid):find("gilded", 1, true)
        if not picked and S.dungeonTries < 4 then
            if now - S.lastDiff > 1.5 then
                S.lastDiff = now
                S.dungeonTries = S.dungeonTries + 1
                log("lobby: dungeon Gilded Skies")
                press(child(cd, "backgroundFillLeft", "ScrollingFrame", "Gilded Skies", "TextButton"), true)
            end
            return true
        end
        if S.diffTries < 2 then
            if now - S.lastDiff > 1.5 then
                S.lastDiff = now
                S.diffTries = S.diffTries + 1
                log("lobby: difficulty " .. CFG.Difficulty)
                press(child(cd, "backgroundFillRight", "difficultyPanel", CFG.Difficulty, "TextButton"), true)
            end
            return true
        end
        return go("create lobby", child(cd, "backgroundFillMiddle", "startMain", "TextButton"), false, 5)
    end
    S.diffTries = 0
    S.dungeonTries = 0
    if sel and isShown(sel) then
        return go("create", child(sel, "Frame", "createGame"), false, 4)
    end
    return go("play", child(pgui(), "mainInterface", "buttons", "playButton"), false, 6)
end

local function startRun()
    S.runStart = os.clock()
    S.runs = S.runs + 1
    writeFile("DQR_runs.txt", S.runs)
    S.runDeaths = 0
    S.lastEnemyT = S.runStart
    S.black = {}
    S.target = nil
    S.roomDone = {}
    S.roomSince = {}
    S.roomIdle = {}
    S.lookUntil = 0
    S.bossPos = nil
    Mover.clear()
    log("run started")
end

local function endRun()
    local elapsed = os.clock() - S.runStart
    S.runStart = nil
    S.waitT = nil
    Mover.clear()
    if elapsed > 40 then
        S.lastClear = elapsed
        S.breakUntil = os.clock() + jitter(CFG.BreakMin, CFG.BreakMax)
        log(string.format("run %d finished in %.0fs", S.runs, elapsed))
        if CFG.MaxRuns > 0 and S.runs >= CFG.MaxRuns then Stop("max runs reached") end
    end
end

local function step()
    local _, hum, hrp = getChar()
    if not hrp then
        S.state = "DEAD / LOADING"
        return
    end
    local now = os.clock()
    local pos = hrp.Position
    if now < S.breakUntil then
        S.state = "BREAK"
        if now >= S.nextJump then
            S.nextJump = now + jitter(25, 70)
            hum.Jump = true
        end
        return
    end
    S.dungeon = (game.PlaceId == CFG.DungeonPlace)
    if inDungeon(pos) then
        local ds = workspace:FindFirstChild("dungeonStarted")
        local st = workspace:FindFirstChild("start")
        local started = true
        if ds and ds:IsA("BoolValue") and not ds.Value then started = false end
        if st and st:IsA("BoolValue") and not st.Value then started = false end
        if not S.runStart and not started then
            S.waitT = S.waitT or now
            if now - S.waitT < 90 then
                S.state = "WAIT START"
                dungeonStartStep(hum, hrp, now)
                return
            end
        end
        if not S.runStart then startRun() end
        local elapsed = now - S.runStart
        local fin = child(workspace, "dungeon", "bossRoom", "dungeonFinished")
        if fin and fin:IsA("BoolValue") and fin.Value then
            S.state = "FINISHED"
            Mover.clear()
            if now >= S.nextEnd then
                S.nextEnd = now + 3
                tryFinish(true)
            end
            return
        end
        if elapsed > CFG.RunTimeout and now >= S.nextEnd then
            S.nextEnd = now + 3
            if not tryFinish(true) and elapsed > CFG.RunTimeout + 150 then
                Stop("run timeout, could not leave the dungeon")
                return
            end
        end
        if now - S.lastEnemyT > 6 and elapsed > 45 and now >= S.nextEnd then
            S.nextEnd = now + 2
            tryFinish(now - S.lastEnemyT > 75 and elapsed > 150 and not roomsPending())
        end
        brain(hum, hrp)
    else
        if S.runStart then endRun() end
        if CFG.MaxRuns > 0 and S.runs >= CFG.MaxRuns then
            Stop("max runs reached")
            return
        end
        if S.resumed and not S.lobbyBreak then
            S.lobbyBreak = true
            S.breakUntil = now + jitter(CFG.BreakMin, CFG.BreakMax)
            return
        end
        S.state = "LOBBY"
        if CFG.AutoEnter then lobbyFlow(now) end
    end
end

local function mainLoop()
    while S.running and alive do
        local ok, err = pcall(step)
        if ok then
            S.errs = 0
        else
            S.errs = S.errs + 1
            log("step error: " .. tostring(err))
            if S.errs > 25 then Stop("too many errors, dump recon and send it") end
        end
        task.wait(jitter(0.07, 0.11))
    end
end

local function queueReload()
    local q = queue_on_teleport or (syn and syn.queue_on_teleport) or (fluxus and fluxus.queue_on_teleport)
    if not q or CFG.ScriptURL == "" then return false end
    local code = string.format("if not game:IsLoaded() then game.Loaded:Wait() end task.wait(2) loadstring(game:HttpGet(%q))()", CFG.ScriptURL)
    local ok = pcall(q, code)
    if ok then S.queued = true end
    return ok
end

local function stopCore(reason, keepFlag)
    local was = S.running
    S.running = false
    S.state = "OFF"
    Mover.clear()
    Aim.release()
    local _, hum, hrp = getChar()
    if hum then
        hum.AutoRotate = true
        if hrp then hum:MoveTo(hrp.Position) end
    end
    if was and not keepFlag then writeFile("DQR_autostart.txt", "0") end
    if was and reason then notify("Autofarm stopped", reason, 8) end
    log("stopped: " .. tostring(reason))
end

function Start()
    if S.running then return end
    S.running = true
    S.state = "STARTING"
    writeFile("DQR_autostart.txt", "1")
    if not queueReload() then
        notify("Heads up", "queue_on_teleport missing: re-run the loader after each teleport.", 8)
    end
    task.spawn(mainLoop)
end

function Stop(reason)
    stopCore(reason, false)
end

local function onDeath()
    if not S.running then return end
    S.deaths = S.deaths + 1
    S.runDeaths = S.runDeaths + 1
    log("died (run deaths " .. S.runDeaths .. ")")
    if S.runDeaths >= CFG.MaxDeathsPerRun then
        Stop("died " .. S.runDeaths .. " times in one run")
    end
end

local function resumeFarm(toggle)

    task.spawn(function()

            task.wait(0.5)
            t = t + 0.5
        end

    end)

local Theme = {

}

local function mk(class, props, parent)
    local o = Instance.new(class)
    if props then
        for k, v in pairs(props) do o[k] = v end
    end
    if parent then o.Parent = parent end
    return o
end

local function round(o, r) return mk("UICorner", { CornerRadius = UDim.new(0, r) }, o) end

local function outline(o, col, th)
    return mk("UIStroke", { Color = col, Thickness = th or 1, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }, o)
end

local function pad(o, l, r, t, b)
    return mk("UIPadding", {
        PaddingLeft = UDim.new(0, l or 0), PaddingRight = UDim.new(0, r or 0),
        PaddingTop = UDim.new(0, t or 0), PaddingBottom = UDim.new(0, b or 0),
    }, o)
end

local function tw(o, t, props)
    if not UI.anim then
        for k, v in pairs(props) do o[k] = v end
        return
    end
    TweenService:Create(o, TweenInfo.new(t, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), props):Play()
end

local function txt(parent, str, size, col, font, xa)
    return mk("TextLabel", {
        BackgroundTransparency = 1, Text = str, TextSize = size, TextColor3 = col, Font = font or Enum.Font.GothamMedium,
        TextXAlignment = xa or Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Center, BorderSizePixel = 0,
    }, parent)
end

local function regSet(flag, get, set)
    UI.reg[flag] = { get = get, set = set }
    UI.regList[#UI.regList + 1] = flag
    local ok, v = pcall(get)
    if ok then UI.defaults[flag] = v end
end

local function snapshotSettings()
    local t = {}
    for _, f in ipairs(UI.regList) do
        local ok, v = pcall(UI.reg[f].get)
        if ok and v ~= nil then t[f] = v end
    end
    return t
end

local function applySettings(t)
    for f, v in pairs(t) do
        local r = UI.reg[f]
        if r then pcall(r.set, v) end
    end
end

function UI.queueSave()
    if UI.saving or not UI.ready then return end
    UI.saving = true
    task.delay(1, function()
        UI.saving = false
        writeFile("DQR_settings.json", HttpService:JSONEncode(snapshotSettings()))
    end)
end

function UI.closeMenu()
    if UI.menu then
        UI.menu:Destroy()
        UI.menu = nil
        UI.menuAnchor = nil
    end
end

function UI.openMenu(anchor, options, current, pick)
    UI.closeMenu()
    local win = UI.win
    local sc = UI.scale.Scale
    local rel = (anchor.AbsolutePosition - win.AbsolutePosition) / sc
    local w = anchor.AbsoluteSize.X / sc
    local h = #options * 26 + 8
    local y = rel.Y + anchor.AbsoluteSize.Y / sc + 4
    if y + h > 534 then y = rel.Y - h - 4 end
    local m = mk("Frame", {
        Size = UDim2.fromOffset(w, h), Position = UDim2.fromOffset(rel.X, y), BackgroundColor3 = Theme.s3,
        BorderSizePixel = 0, ZIndex = 60,
    }, win)
    round(m, 8)
    outline(m, Theme.ln2)
    pad(m, 4, 4, 4, 4)
    mk("UIListLayout", { SortOrder = Enum.SortOrder.LayoutOrder }, m)
    for i, opt in ipairs(options) do
        local b = mk("TextButton", {
            Size = UDim2.new(1, 0, 0, 26), BackgroundColor3 = Theme.em, BackgroundTransparency = 1, Text = tostring(opt),
            TextSize = 12, Font = Enum.Font.GothamMedium, TextColor3 = (opt == current) and Theme.em or Theme.tx,
            ZIndex = 61, LayoutOrder = i, AutoButtonColor = false, BorderSizePixel = 0,
        }, m)
        round(b, 6)
        b.MouseEnter:Connect(function() b.BackgroundTransparency = 0.85 end)
        b.MouseLeave:Connect(function() b.BackgroundTransparency = 1 end)
        b.MouseButton1Click:Connect(function()
            UI.closeMenu()
            pick(opt)
        end)
    end
    UI.menu = m
    UI.menuAnchor = anchor
end

local function newRow(card, o, h)
    card.n = card.n + 1
    local row = mk("Frame", { Size = UDim2.new(1, 0, 0, h), BackgroundTransparency = 1, LayoutOrder = card.n, BorderSizePixel = 0 }, card.frame)
    if card.n > 2 then
        mk("Frame", { Size = UDim2.new(1, 0, 0, 1), BackgroundColor3 = Theme.white, BackgroundTransparency = 0.965, BorderSizePixel = 0 }, row)
    end
    local nameL = txt(row, o.name, 12, Theme.tx, Enum.Font.GothamMedium)
    local descL
    if o.desc then
        nameL.Position = UDim2.fromOffset(0, 7)
        nameL.Size = UDim2.new(0.6, 0, 0, 16)
        descL = txt(row, o.desc, 10.5, Theme.mu, Enum.Font.Gotham)
        descL.Position = UDim2.fromOffset(0, 24)
        descL.Size = UDim2.new(0.62, 0, 0, 14)
    else
        nameL.Position = UDim2.fromOffset(0, 0)
        nameL.Size = UDim2.new(0.6, 0, 1, 0)
    end
    UI.rows[#UI.rows + 1] = { nameL = nameL, descL = descL, key = (o.name .. " " .. (o.desc or "")):lower() }
    return row, nameL
end

function Card:Toggle(o)
    local row = newRow(self, o, o.desc and 46 or 36)
    local track = mk("TextButton", {
        Text = "", AutoButtonColor = false, Size = UDim2.fromOffset(38, 20), AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, 0, 0.5, 0), BackgroundColor3 = Theme.s3, BorderSizePixel = 0,
    }, row)
    round(track, 10)
    local st = outline(track, Theme.ln2)
    local knob = mk("Frame", {
        Size = UDim2.fromOffset(14, 14), AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0, 3, 0.5, 0),
        BackgroundColor3 = Theme.mu, BorderSizePixel = 0,
    }, track)
    round(knob, 7)
    local w = { value = false }
    function w.Set(v, silent)
        w.value = v and true or false
        tw(track, 0.2, { BackgroundColor3 = w.value and Theme.em2 or Theme.s3 })
        tw(knob, 0.2, {
            Position = w.value and UDim2.new(0, 21, 0.5, 0) or UDim2.new(0, 3, 0.5, 0),
            BackgroundColor3 = w.value and Theme.white or Theme.mu,
        })
        st.Color = w.value and Theme.em2 or Theme.ln2
        if not silent then
            if o.set then o.set(w.value) end
            UI.queueSave()
        end
    end
    track.MouseButton1Click:Connect(function() w.Set(not w.value) end)
    if o.get then w.Set(o.get(), true) end
    if o.flag then regSet(o.flag, o.get, function(v) w.Set(v) end) end
    return w
end

function Card:Slider(o)
    self.n = self.n + 1
    local row = mk("Frame", { Size = UDim2.new(1, 0, 0, 56), BackgroundTransparency = 1, LayoutOrder = self.n, BorderSizePixel = 0 }, self.frame)
    if self.n > 2 then
        mk("Frame", { Size = UDim2.new(1, 0, 0, 1), BackgroundColor3 = Theme.white, BackgroundTransparency = 0.965, BorderSizePixel = 0 }, row)
    end
    local nameL = txt(row, o.name, 12, Theme.tx, Enum.Font.GothamMedium)
    nameL.Position = UDim2.fromOffset(0, 8)
    nameL.Size = UDim2.new(0.65, 0, 0, 16)
    local valL = txt(row, "", 11, Theme.em, Enum.Font.Code, Enum.TextXAlignment.Right)
    valL.Position = UDim2.new(0.5, 0, 0, 8)
    valL.Size = UDim2.new(0.5, 0, 0, 16)
    UI.rows[#UI.rows + 1] = { nameL = nameL, valL = valL, key = (o.name .. " " .. (o.desc or "")):lower() }
    local hit = mk("TextButton", {
        Text = "", AutoButtonColor = false, BackgroundTransparency = 1, Position = UDim2.fromOffset(0, 30), Size = UDim2.new(1, 0, 0, 20),
        BorderSizePixel = 0,
    }, row)
    local bar = mk("Frame", { Size = UDim2.new(1, 0, 0, 5), Position = UDim2.new(0, 0, 0.5, -2), BackgroundColor3 = Theme.s3, BorderSizePixel = 0 }, hit)
    round(bar, 3)
    local fill = mk("Frame", { Size = UDim2.new(0, 0, 1, 0), BackgroundColor3 = Theme.em2, BorderSizePixel = 0 }, bar)
    round(fill, 3)
    local knob = mk("Frame", {
        Size = UDim2.fromOffset(14, 14), AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0, 0, 0.5, 0),
        BackgroundColor3 = Theme.white, BorderSizePixel = 0, ZIndex = 3,
    }, bar)
    round(knob, 7)
    outline(knob, Theme.em, 2)
    local step = o.step or 1
    local decimals = 0
    if step < 1 then decimals = (step >= 0.1) and 1 or 2 end
    local w = { value = o.min }
    function w.Set(v, silent)
        v = tonumber(v) or o.min
        v = math.clamp(v, o.min, o.max)
        v = o.min + math.floor((v - o.min) / step + 0.5) * step
        v = math.clamp(v, o.min, o.max)
        w.value = v
        local r = (o.max > o.min) and (v - o.min) / (o.max - o.min) or 0
        fill.Size = UDim2.new(r, 0, 1, 0)
        knob.Position = UDim2.new(r, 0, 0.5, 0)
        valL.Text = string.format("%." .. decimals .. "f", v) .. (o.suffix or "")
        if not silent then
            if o.set then o.set(v) end
            UI.queueSave()
        end
    end
    local function fromPos(p)
        local r = math.clamp((p.X - hit.AbsolutePosition.X) / math.max(hit.AbsoluteSize.X, 1), 0, 1)
        w.Set(o.min + r * (o.max - o.min))
    end
    hit.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            UI.drag = fromPos
            fromPos(i.Position)
        end
    end)
    if o.get then w.Set(o.get(), true) end
    if o.flag then regSet(o.flag, o.get, function(v) w.Set(v) end) end
    return w
end

local B64CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local B64MAP = {}
for i = 1, #B64CHARS do B64MAP[B64CHARS:sub(i, i)] = i - 1 end

local function b64decode(str)
    local out = {}
    local acc, bits = 0, 0
    for i = 1, #str do
        local v = B64MAP[str:sub(i, i)]
        if v then
            acc = acc * 64 + v
            bits = bits + 6
            if bits >= 8 then
                bits = bits - 8
                local byte = math.floor(acc / 2 ^ bits)
                acc = acc - byte * 2 ^ bits
                out[#out + 1] = string.char(byte)
            end
        end
    end
    return table.concat(out)
end

local function loadAsset(fileName, chunkList)
    local getAsset = getcustomasset or getsynasset
    if not getAsset or not writefile then return nil end
    local ok, res = pcall(function()
        ensureFolder()
        local path = CFG.Folder .. "/" .. fileName
        writefile(path, b64decode(table.concat(chunkList)))
        return getAsset(path)
    end)
    if ok and res and res ~= "" then return res end
    return nil
end

local function fmtTime(sec)
    sec = math.max(0, math.floor(sec))
    return string.format("%02d:%02d:%02d", math.floor(sec / 3600), math.floor(sec / 60) % 60, sec % 60)
end

local function farmSet(v)
    if v then
        if not S.resumed then
            S.runs = 0
            writeFile("DQR_runs.txt", 0)
        end
        Start()
    else
        Stop()
    end
end

local function applyPerf()
    pcall(function() settings().Rendering.QualityLevel = CFG.LowQuality and Enum.QualityLevel.Level01 or Enum.QualityLevel.Automatic end)
    pcall(function() game:GetService("Lighting").GlobalShadows = not CFG.NoShadows end)
    pcall(function() RunService:Set3dRenderingEnabled(not CFG.No3D) end)
    if setfpscap then pcall(setfpscap, CFG.FpsCapOn and CFG.FpsCap or 0) end
end

    local raw = readFile("DQR_cfgindex.json")

        toast.Size = UDim2.fromOffset(math.clamp(#msg * 6.6 + 40, 180, 700), 34)
        toast.Visible = true
        toast.Position = UDim2.new(0.5, 0, 1, 50)
        tw(toast, 0.4, { Position = UDim2.new(0.5, 0, 1, -22) })
        UI.toastId = (UI.toastId or 0) + 1
        local id = UI.toastId
        task.delay(dur or 4, function()
            if UI.toastId == id and toast.Parent then
                tw(toast, 0.4, { Position = UDim2.new(0.5, 0, 1, 50) })
                task.delay(0.5, function()
                    if UI.toastId == id and toast.Parent then toast.Visible = false end
                end)
            end
        end)
    end

    track(UIS.InputBegan:Connect(function(i, gp)
        if UI.menu and (i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch) then
            local p = UIS:GetMouseLocation()
            local function inside(o)
                if not o or not o.Parent then return false end
                local a, s = o.AbsolutePosition, o.AbsoluteSize
                return p.X >= a.X and p.X <= a.X + s.X and p.Y >= a.Y and p.Y <= a.Y + s.Y
            end
            if not inside(UI.menu) and not inside(UI.menuAnchor) then UI.closeMenu() end
        end
        if gp or UI.capturing or i.UserInputType ~= Enum.UserInputType.Keyboard then return end
        if i.KeyCode == UI.keys.menu then
            UI.setVisible(not win.Visible)
        elseif i.KeyCode == UI.keys.farm then
            local on = not S.running
            farmSet(on)
            if UI.farmW then UI.farmW.Set(on, true) end
        elseif i.KeyCode == UI.keys.panic then
            if S.running then
                Stop("panic key")
                if UI.farmW then UI.farmW.Set(false, true) end
            end
        end
    end))

    task.spawn(function()
        while alive do
            for _, u in ipairs(UI.updaters) do pcall(u) end
            if UI.farmW and UI.farmW.value ~= S.running then UI.farmW.Set(S.running, true) end
            task.wait(1)
        end
    end)

    uiDestroy = function()
        pcall(function() gui:Destroy() end)
    end

    UI.showTab("home")

    local raw = readFile("DQR_settings.json")
    if raw then
        local ok, t = pcall(function() return HttpService:JSONDecode(raw) end)
        if ok and type(t) == "table" then applySettings(t) end
    end
    local auto = readFile("DQR_autoload.txt")
    if auto and auto ~= "" then
        local craw = readFile(cfgFile(auto))
        if craw then
            local ok, t = pcall(function() return HttpService:JSONDecode(craw) end)
            if ok and type(t) == "table" then applySettings(t) end
        end
    end
    applyPerf()
    UI.ready = true
    return true
end

    log("UI failed: " .. tostring(uiErr))
